import { buildArtworkMetadata, truncateDescription } from '@/lib/gallery/metadata-utils'

const mockArtwork = {

  title: 'Portrait au crayon',
  description: 'Une étude soignée du visage humain.',
  widthCm: '30',

    expect(truncateDescription(long)).toHaveLength(160)
  })

  it('retourne undefined si la description est null', () => {
    expect(truncateDescription(null)).toBeUndefined()
  })

  it('retourne undefined si la description est undefined', () => {
    expect(truncateDescription(undefined)).toBeUndefined()
  })

describe('buildArtworkMetadata', () => {
  it('construit le titre au format "{titre} — Galerie POF"', () => {
    const meta = buildArtworkMetadata(mockArtwork, storageBaseUrl)
    expect(meta.title).toBe('Portrait au crayon — Galerie POF')
  })

  it('utilise la description de l\'œuvre tronquée à 160 chars', () => {
    const meta = buildArtworkMetadata(mockArtwork, storageBaseUrl)
    expect(meta.openGraph?.description).toBe('Une étude soignée du visage humain.')
  })

  it('génère une description fallback si pas de description', () => {
    const artwork = { ...mockArtwork, description: null }
    const meta = buildArtworkMetadata(artwork, storageBaseUrl)
    expect(meta.openGraph?.description).toContain('Dessin')
    expect(meta.openGraph?.description).toContain('Crayon graphite')
    expect(meta.openGraph?.description).toContain('30×40 cm')
  })

  it('inclut l\'URL d\'image dans og:images si processedPath présent', () => {
    const meta = buildArtworkMetadata(mockArtwork, storageBaseUrl)
    expect(meta.openGraph?.images).toHaveLength(1)
    expect((meta.openGraph?.images as string[])[0]).toContain('processed/abc123.webp')
  })

  it('retourne og:images vide si pas de processedPath', () => {
    const artwork = { ...mockArtwork, processedPath: null }
    const meta = buildArtworkMetadata(artwork, storageBaseUrl)
    expect(meta.openGraph?.images).toHaveLength(0)
  })

/**
 * Fonctions utilitaires pour la construction des métadonnées Open Graph
 * des pages détail d'œuvres.
 *
 * POURQUOI extraire hors du composant de page ?
 * Testabilité unitaire sans instancier Next.js ni accéder à la DB.
 * La fonction buildArtworkMetadata est purement fonctionnelle.
 */
import type { Metadata } from 'next'

interface ArtworkForMetadata {
  title: string
  description: string | null | undefined
  widthCm: string | null | undefined
  heightCm: string | null | undefined
  processedPath: string | null | undefined
  type: { name: string } | null
  medium: { name: string } | null
}

/**
 * Tronque une description à 160 caractères (limite og:description).
 * Retourne undefined si la valeur est null/undefined.
 */
export function truncateDescription(
  description: string | null | undefined
): string | undefined {
  if (!description) return undefined
  return description.slice(0, 160)
}

/**
 * Construit l'objet Metadata Next.js pour une page détail d'œuvre.
 *
 * @param artwork - Données de l'œuvre avec ses relations type et médium
 * @param imageBaseUrl - URL de base pour construire l'URL complète de l'image
 *                       (retourné par storage.getUrl() côté serveur)
 */
export function buildArtworkMetadata(
  artwork: ArtworkForMetadata,
  imageBaseUrl: string
): Metadata {
  // Description : utilise la description de l'œuvre ou génère un fallback
  const description =
    truncateDescription(artwork.description) ??
    `${artwork.type?.name ?? ''} · ${artwork.medium?.name ?? ''} · ${artwork.widthCm ?? '?'}×${artwork.heightCm ?? '?'} cm`

  // URL de l'image og:image — doit être absolue pour les crawlers sociaux (Piège 1 RESEARCH.md)
  const imageUrl = artwork.processedPath ? imageBaseUrl : null

  return {
    title: `${artwork.title} — Galerie POF`,
    openGraph: {
      title: artwork.title,
      description,
      images: imageUrl ? [imageUrl] : [],
      type: 'website',
    },
  }
}

  next: ArtworkNavItem | null // Plus ancien (affiché à droite)

/**
 * Génère les métadonnées Open Graph pour le partage social.
 * Appelé séparément de la page par Next.js.
 *
 * IMPORTANT : metadataBase est défini dans layout.tsx →
 * les URLs relatives sont automatiquement résolues en absolues.

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { id } = await params
  const artworkId = parseInt(id, 10)
  if (isNaN(artworkId)) return { title: '404 — Galerie POF' }

  const artwork = await db.query.artworks.findFirst({
    where: (a, { and, eq }) => and(
      eq(a.id, artworkId),
      // SÉCURITÉ : ne jamais exposer les brouillons via og:image
      eq(a.pipelineState, 'published')
    ),
    with: { type: true, medium: true },
  })

  if (!artwork) return { title: '404 — Galerie POF' }

  const storage = getStorage()
  const imageUrl = artwork.processedPath ? storage.getUrl(artwork.processedPath) : undefined

  const description =
    artwork.description?.slice(0, 160) ??
    `${artwork.type?.name ?? ''} · ${artwork.medium?.name ?? ''} · ${artwork.widthCm ?? '?'}×${artwork.heightCm ?? '?'} cm`

  return {
    title: `${artwork.title} — Galerie POF`,
    openGraph: {
      title: artwork.title,
      description,
      images: imageUrl ? [imageUrl] : [],
      type: 'website',
    },
  }
}
