/**
 * Fonctions helper pour la galerie publique.
 *
 * POURQUOI ces helpers ?
 * Drizzle retourne les colonnes `numeric` PostgreSQL comme des strings JavaScript.
 * (Comportement du driver node-postgres — voir piège 4 dans 03-RESEARCH.md)
 * Ces fonctions convertissent de manière sûre pour l'aspect-ratio CSS.
 */

/**
 * Convertit une valeur numeric Drizzle (string) en number.
 * Retourne 1 comme fallback pour éviter les divisions par zéro.
 */
export function parseDimension(value: string | null | undefined): number {
  if (!value) return 1
  const n = parseFloat(value)
  return isNaN(n) || n === 0 ? 1 : n
}

/**
 * Construit la string CSS aspect-ratio depuis les dimensions en cm.
 * Exemple : widthCm="30", heightCm="20" → "30 / 20"
 */
export function buildAspectRatio(
  widthCm: string | null | undefined,
  heightCm: string | null | undefined
): string {
  const w = parseDimension(widthCm)
  const h = parseDimension(heightCm)
  if (w === 1 && !widthCm) return '1 / 1'
  if (h === 1 && !heightCm) return '1 / 1'
  return `${w} / ${h}`
}

/**

/**
 * Déclencheur de chargement pour l'infinite scroll.
 *
 * Un <div> invisible de 1px placé après le dernier lot d'œuvres.
 * L'IntersectionObserver le surveille et déclenche le chargement
 * du lot suivant quand il entre dans le viewport.
 *
 * POURQUOI ce composant séparé ?
 * Isole la logique IntersectionObserver du composant GalleryClient
 * pour la testabilité. Reçoit une callback onIntersect en prop.
 */
'use client'
import { useEffect, useRef } from 'react'

interface ScrollSentinelProps {
  onIntersect: () => void
  hasMore: boolean
}

export function ScrollSentinel({ onIntersect, hasMore }: ScrollSentinelProps) {
  const sentinelRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    if (!hasMore) return // Ne pas observer si toutes les œuvres sont affichées

    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting) {
          onIntersect()
        }
      },
      { threshold: 0.1 }
    )

    if (sentinelRef.current) observer.observe(sentinelRef.current)

    // Cleanup OBLIGATOIRE : évite les fuites mémoire si le composant est démonté
    return () => observer.disconnect()
  }, [onIntersect, hasMore])

  if (!hasMore) return null

  return (
    <div ref={sentinelRef} style={{ height: '1px', width: '100%' }}>
      {/* Indicateur de chargement : 3 carrés pulse teal (UI-SPEC section 7) */}
      <div
        style={{
          display: 'flex',
          justifyContent: 'center',
          gap: '4px',
          paddingTop: '24px',
        }}
      >
        {[0, 1, 2].map((i) => (
          <div
            key={i}
            style={{
              width: '8px',
              height: '8px',
              backgroundColor: 'var(--color-teal)',
              animation: `pulse 1.2s ease-in-out ${i * 0.2}s infinite`,
            }}
          />
        ))}
      </div>
      <style>{`
        @keyframes pulse {
          0%, 100% { opacity: 0.2; }
          50% { opacity: 1; }
        }
      `}</style>
    </div>
  )
}

/**
 * Composant client principal de la galerie.
 *
 * POURQUOI 'use client' ?
 * Ce composant utilise :
 * 1. useSearchParams — lire les filtres depuis l'URL
 * 2. useRouter — mettre à jour l'URL sans rechargement de page
 * 3. useState — visibleCount, pas besoin d'autre état
 * 4. useEffect — réinitialiser visibleCount lors d'un changement de filtre
 *
 * ARCHITECTURE (03-RESEARCH.md Pattern 1) :
 * Le Server Component parent (page.tsx) charge TOUTES les œuvres publiées
 * une seule fois, les passe en props ici. On filtre et on pagine côté client.
 * Viable pour une collection < 500 œuvres — voir hypothèse A1 dans RESEARCH.md.
 *
 * PIÈGE 2 : useSearchParams doit être dans un composant enveloppé par <Suspense>
 * dans le Server Component parent. Sinon next build échoue.

'use client'
import { useState, useEffect } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { MasonryGrid } from './MasonryGrid'
import { ArtworkCard } from './ArtworkCard'
import { ScrollSentinel } from './ScrollSentinel'
import { FilterBar } from './FilterBar'
import type { ArtworkType, Medium } from '@/db/schema'

// Nombre d'œuvres chargées par lot (D-08)
const BATCH_SIZE = 6

interface ArtworkWithUrls {
  id: number
  title: string
  status: string
  thumbnailUrl: string | null
  processedUrl: string | null
  widthCmNum: number
  heightCmNum: number
  typeId: number | null
  mediumId: number | null
}

interface GalleryClientProps {
  artworks: ArtworkWithUrls[]
  types: ArtworkType[]
  mediums: Medium[]
}

export function GalleryClient({ artworks, types, mediums }: GalleryClientProps) {
  const router = useRouter()
  const searchParams = useSearchParams()

  // Lire les filtres depuis l'URL (persistance D-13)
  const activeTypeId = searchParams.get('type') ?? ''
  const activeMediumId = searchParams.get('medium') ?? ''

  const [visibleCount, setVisibleCount] = useState(BATCH_SIZE)

  // CRITIQUE (Piège 6) : réinitialiser le compteur à chaque changement de filtre
  // Sans ce reset, on peut afficher 0 résultat si les N premiers items
  // ne correspondent pas au filtre actif.
  useEffect(() => {
    setVisibleCount(BATCH_SIZE)
  }, [activeTypeId, activeMediumId])

  // Filtrage côté client — appliqué sur les données complètes
  const filteredArtworks = artworks.filter((a) => {
    if (activeTypeId && String(a.typeId) !== activeTypeId) return false
    if (activeMediumId && String(a.mediumId) !== activeMediumId) return false
    return true
  })

  const visibleArtworks = filteredArtworks.slice(0, visibleCount)
  const hasMore = visibleCount < filteredArtworks.length

  function loadMore() {
    setVisibleCount((n) => Math.min(n + BATCH_SIZE, filteredArtworks.length))
  }

  function setFilter(key: 'type' | 'medium', value: string) {
    const params = new URLSearchParams(searchParams.toString())
    if (value) {
      params.set(key, value)
    } else {
      params.delete(key)
    }
    // scroll: false évite le saut en haut de page à chaque clic de filtre
    router.replace(`/?${params.toString()}`, { scroll: false })
  }

  return (
    <div style={{ position: 'relative', zIndex: 1 }}>
      {/* Barre de filtres — sticky sous la nav (D-12, D-13) */}
      <FilterBar
        types={types}
        mediums={mediums}
        activeTypeId={activeTypeId}
        activeMediumId={activeMediumId}
        onTypeChange={(id) => setFilter('type', id)}
        onMediumChange={(id) => setFilter('medium', id)}
      />

      {/* État vide — UI-SPEC section 12 */}
      {filteredArtworks.length === 0 && (
        <div style={{ textAlign: 'center', padding: '64px 24px' }}>
          <p
            style={{
              fontFamily: 'var(--font-oswald)',
              fontSize: '20px',
              fontWeight: 700,
              color: 'var(--color-ink)',
              marginBottom: '8px',
            }}
          >
            {activeTypeId || activeMediumId
              ? 'Aucun résultat pour ces filtres. Essayez une autre combinaison.'
              : 'Aucune œuvre publiée pour l\'instant.'}
          </p>
          {!activeTypeId && !activeMediumId && (
            <p style={{ fontFamily: 'var(--font-space-mono)', fontSize: '11px', color: 'var(--color-ink-muted)' }}>
              Revenez bientôt.
            </p>
          )}
        </div>
      )}

      {/* Grille masonry */}
      {visibleArtworks.length > 0 && (
        <MasonryGrid>
          {visibleArtworks.map((artwork) => (
            <ArtworkCard key={artwork.id} {...artwork} />
          ))}
        </MasonryGrid>
      )}

      {/* Sentinel invisible — déclenche le chargement du lot suivant */}
      <ScrollSentinel onIntersect={loadMore} hasMore={hasMore} />
    </div>
  )
}
