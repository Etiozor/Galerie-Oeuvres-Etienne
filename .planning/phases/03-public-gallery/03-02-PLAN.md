```typescript
/**
 * Tests unitaires pour la logique de filtrage de la galerie.
 *
 * Ces tests couvrent :
 * 1. filterArtworks : filtrage d'une liste d'œuvres par type et médium
 * 2. buildFilterUrl : construction des URL params pour la persistance des filtres

 * La logique est extraite dans src/lib/gallery/filter-utils.ts pour

import { describe, it, expect } from 'vitest'
import { filterArtworks, buildFilterUrl } from '@/lib/gallery/filter-utils'

// Fixtures d'œuvres pour les tests
const mockArtworks = [
  { id: 1, typeId: 1, mediumId: 1, title: 'Dessin crayon A' },
  { id: 2, typeId: 1, mediumId: 2, title: 'Dessin encre B' },
  { id: 3, typeId: 2, mediumId: 1, title: 'Peinture acrylique C' },
  { id: 4, typeId: 2, mediumId: 3, title: 'Peinture huile D' },
  { id: 5, typeId: null, mediumId: null, title: 'Sans classification' },
]

describe('filterArtworks', () => {
  it('retourne tous les items si aucun filtre actif', () => {
    expect(filterArtworks(mockArtworks, '', '')).toHaveLength(5)
  })

  it('filtre uniquement par type', () => {
    const result = filterArtworks(mockArtworks, '1', '')
    expect(result).toHaveLength(2)
    expect(result.map((a) => a.id)).toEqual([1, 2])
  })

  it('filtre uniquement par médium', () => {
    const result = filterArtworks(mockArtworks, '', '1')
    expect(result).toHaveLength(2)
    expect(result.map((a) => a.id)).toEqual([1, 3])
  })

  it('filtre par type ET médium (intersection)', () => {
    const result = filterArtworks(mockArtworks, '1', '2')
    expect(result).toHaveLength(1)
    expect(result[0].id).toBe(2)
  })

  it("retourne [] si aucune œuvre ne correspond au filtre", () => {
    expect(filterArtworks(mockArtworks, '99', '')).toHaveLength(0)
  })

  it('retourne [] pour une liste vide', () => {
    expect(filterArtworks([], '1', '')).toHaveLength(0)
  })

  it('exclut les œuvres sans typeId quand un filtre type est actif', () => {
    const result = filterArtworks(mockArtworks, '1', '')
    expect(result.map((a) => a.id)).not.toContain(5)
  })
})

describe('buildFilterUrl', () => {
  it('ajoute un paramètre de filtre à des params vides', () => {
    const params = new URLSearchParams()
    expect(buildFilterUrl('type', '1', params)).toBe('type=1')
  })

  it('supprime un paramètre quand la valeur est vide', () => {
    const params = new URLSearchParams('type=1&medium=2')
    expect(buildFilterUrl('type', '', params)).toBe('medium=2')
  })

  it('remplace un paramètre existant', () => {
    const params = new URLSearchParams('type=1')
    expect(buildFilterUrl('type', '2', params)).toBe('type=2')
  })

  it('conserve les autres params lors de la suppression', () => {
    const params = new URLSearchParams('type=1&medium=3')
    const result = buildFilterUrl('medium', '', params)
    expect(result).toBe('type=1')
  })
})
```

**2. Créer `src/lib/gallery/filter-utils.ts`** — extraction de la logique de filtrage :

```typescript
/**
 * Fonctions utilitaires pour le filtrage de la galerie.
 *
 * POURQUOI extraire la logique hors du composant React ?
 * Testabilité : ces fonctions sont purement fonctionnelles (input → output)
 * et peuvent être testées avec Vitest sans monter de composant React.
 *
 * filterArtworks est utilisée par GalleryClient pour filtrer les œuvres
 * côté client sur les données chargées une seule fois côté serveur (Pattern 1 RESEARCH.md).
 */

interface FilterableArtwork {
  typeId: number | null
  mediumId: number | null
  [key: string]: unknown
}

/**
 * Filtre une liste d'œuvres par type et/ou médium.
 *
 * @param artworks - Liste complète des œuvres
 * @param activeTypeId - ID du type actif comme string (ex: "1"), "" si aucun
 * @param activeMediumId - ID du médium actif comme string, "" si aucun
 * @returns Sous-liste des œuvres correspondant aux filtres
 */
export function filterArtworks<T extends FilterableArtwork>(
  artworks: T[],
  activeTypeId: string,
  activeMediumId: string
): T[] {
  return artworks.filter((a) => {
    // Filtre type : si un type est sélectionné, l'œuvre doit correspondre
    if (activeTypeId && String(a.typeId) !== activeTypeId) return false
    // Filtre médium : si un médium est sélectionné, l'œuvre doit correspondre
    if (activeMediumId && String(a.mediumId) !== activeMediumId) return false
    return true
  })
}

/**
 * Construit la chaîne de query string pour la mise à jour de l'URL.
 *
 * Prend les URLSearchParams courants, ajoute ou supprime le paramètre
 * spécifié selon si la valeur est non-vide ou vide.
 *
 * @param key - Clé du paramètre ('type' ou 'medium')
 * @param value - Nouvelle valeur, "" pour supprimer
 * @param currentParams - URLSearchParams actuels (depuis useSearchParams)
 * @returns Nouvelle chaîne de query string (sans le '?')
 */
export function buildFilterUrl(
  key: string,
  value: string,
  currentParams: URLSearchParams
): string {
  // IMPORTANT : créer une COPIE pour ne pas muter les params originaux
  const params = new URLSearchParams(currentParams.toString())
  if (value) {
    params.set(key, value)
  } else {
    params.delete(key)
  }
  return params.toString()
}
```

```typescript
'use client'
import type { ArtworkType, Medium } from '@/db/schema'

interface FilterBarProps {
  types: ArtworkType[]
  mediums: Medium[]
  activeTypeId: string
  activeMediumId: string
  onTypeChange: (id: string) => void
  onMediumChange: (id: string) => void
}

/**
 * Styles des boutons de filtre — D-03 zéro coins arrondis, D-UI-SPEC section 3.
 * Utilise des styles inline pour garantir la conformité spec (pas de risque Tailwind purge).
 */
function filterButtonStyle(isActive: boolean): React.CSSProperties {
  return {
    fontFamily: 'var(--font-space-mono)',
    fontSize: '11px',
    fontWeight: 400,
    textTransform: 'uppercase' as const,
    letterSpacing: '0.08em',
    padding: '0 12px',
    height: '28px',
    border: isActive ? '1px solid #1a1a1a' : '1px solid #5a5244',
    backgroundColor: isActive ? '#f4e285' : 'transparent', // Jaune si actif (D-UI-SPEC)
    color: isActive ? '#1a1a1a' : '#5a5244',
    cursor: 'pointer',
    transition: 'background-color 0.15s, border-color 0.15s, color 0.15s',
    // D-03 : borderRadius absent = 0 par défaut
  }
}

export function FilterBar({
  types,
  mediums,
  activeTypeId,
  activeMediumId,
  onTypeChange,
  onMediumChange,
}: FilterBarProps) {
  return (
    <div
      style={{
        position: 'sticky',
        top: '56px', // Hauteur exacte de SiteNav
        zIndex: 9,   // Sous SiteNav (z-index 10)
        backgroundColor: 'var(--color-surface)',
        height: '44px', // Touch target minimum (UI-SPEC)
        display: 'flex',
        alignItems: 'center',
        padding: '0 8px',
        gap: '8px',
        overflowX: 'auto', // Scroll horizontal si trop de médiums sur mobile
        // scrollbar-width: none sur mobile pour éviter la scrollbar visible
      }}
    >
      {/* Séparateur visuel */}
      <span
        style={{
          fontFamily: 'var(--font-space-mono)',
          fontSize: '10px',
          color: 'var(--color-ink-muted)',
          letterSpacing: '0.05em',
          flexShrink: 0,
          marginRight: '4px',
        }}
      >
        TYPE :
      </span>

      {/* Boutons filtres par type (Dessin / Peinture / ...) */}
      {types.map((type) => {
        const isActive = activeTypeId === String(type.id)
        return (
          <button
            key={type.id}
            onClick={() => onTypeChange(isActive ? '' : String(type.id))}
            style={filterButtonStyle(isActive)}
            aria-pressed={isActive}
            aria-label={`Filtrer par ${type.name}`}
          >
            {type.name}
          </button>
        )
      })}

      {/* Séparateur entre les deux groupes de filtres */}
      <span
        style={{
          width: '1px',
          height: '20px',
          backgroundColor: 'var(--color-ink-muted)',
          opacity: 0.3,
          flexShrink: 0,
          marginLeft: '4px',
          marginRight: '4px',
        }}
      />

      <span
        style={{
          fontFamily: 'var(--font-space-mono)',
          fontSize: '10px',
          color: 'var(--color-ink-muted)',
          letterSpacing: '0.05em',
          flexShrink: 0,
          marginRight: '4px',
        }}
      >
        MÉDIUM :
      </span>

      {/* Boutons filtres par médium (Crayon / Encre / Acrylique / ...) */}
      {mediums.map((medium) => {
        const isActive = activeMediumId === String(medium.id)
        return (
          <button
            key={medium.id}
            onClick={() => onMediumChange(isActive ? '' : String(medium.id))}
            style={filterButtonStyle(isActive)}
            aria-pressed={isActive}
            aria-label={`Filtrer par médium ${medium.name}`}
          >
            {medium.name}
          </button>
        )
      })}
    </div>
  )
}
```
