/**
 * Interface d'abstraction du stockage de fichiers — Galerie POF.
 *
 * POURQUOI une interface et pas un accès direct au système de fichiers ?
 * En développement, les images sont stockées sur le disque local.
 * En production (Phase 6), elles seront sur Cloudflare R2 (compatible S3).
 * Cette interface permet de changer d'environnement en modifiant UNE SEULE
 * variable d'environnement (STORAGE_BACKEND), sans toucher au code qui upload
 * ou affiche les images.
 *
 * Convention de clé :
 * La base de données stocke des CLÉS, pas des chemins absolus.
 * Exemples de clés : "originals/abc123.jpg", "thumbnails/abc123.webp"
 * Le StorageProvider traduit la clé en chemin disque ou URL R2.
 * [M6 — PITFALLS.md]
 */

import type { LocalStorageAdapter } from './local'

// ============================================================
// INTERFACE — Contrat que tous les adaptateurs doivent respecter
// ============================================================

/**
 * Interface StorageProvider.
 *
 * Tout adaptateur de stockage (local, R2, S3...) doit implémenter
 * ces 4 méthodes. Le code applicatif n'utilise jamais un adaptateur
 * directement — il passe toujours par getStorage().
 */
export interface StorageProvider {
  /**
   * Stocker un fichier.
   *
   * @param key      Clé de stockage (ex: "originals/abc123.jpg")
   * @param buffer   Contenu binaire du fichier
   * @param options  Options facultatives : type MIME du fichier
   */
  put(
    key: string,
    buffer: Buffer,
    options?: { contentType?: string }
  ): Promise<void>

  /**
   * Récupérer un fichier par sa clé.
   *
   * @param key  Clé de stockage
   * @returns    Buffer contenant les données du fichier
   * @throws     Error si le fichier n'existe pas
   */
  get(key: string): Promise<Buffer>

  /**
   * Supprimer un fichier.
   * Ne lève pas d'erreur si le fichier est absent.
   *
   * @param key  Clé de stockage
   */
  delete(key: string): Promise<void>

  /**
   * Obtenir l'URL publique d'un fichier.
   *
   * En développement local → URL comme "http://localhost:3000/uploads/originals/abc123.jpg"
   * En production R2 → URL CDN comme "https://cdn.galeriepoF.fr/originals/abc123.jpg"
   *
   * @param key  Clé de stockage
   * @returns    URL complète pour afficher le fichier dans le navigateur
   */
  getUrl(key: string): string
}

// ============================================================
// FACTORY — Choisit l'adaptateur selon la variable d'environnement
// ============================================================

/**
 * Factory getStorage() — Retourne l'adaptateur de stockage configuré.
 *
 * Lit la variable STORAGE_BACKEND pour choisir l'implémentation :
 * - "local" → LocalStorageAdapter (disque local, développement)
 * - "r2"    → R2StorageAdapter (Cloudflare R2, Phase 6)
 *
 * IMPORTANT : Toujours utiliser getStorage() plutôt qu'importer
 * LocalStorageAdapter directement. Cela garantit que le swap
 * dev→prod ne nécessite aucun changement de code.
 *
 * @returns Instance de StorageProvider correspondant à STORAGE_BACKEND
 * @throws  Error si STORAGE_BACKEND contient une valeur non reconnue
 */
export function getStorage(): StorageProvider {
  const backend = process.env.STORAGE_BACKEND ?? 'local'

  switch (backend) {
    case 'local': {
      // Import dynamique pour éviter de charger le module fs en production R2
      // eslint-disable-next-line @typescript-eslint/no-require-imports
      const { LocalStorageAdapter } = require('./local') as { LocalStorageAdapter: new () => LocalStorageAdapter }
      return new LocalStorageAdapter()
    }

    case 'r2':
      // R2StorageAdapter sera implémenté en Phase 6
      // Pour l'instant, lever une erreur explicite pour éviter les bugs silencieux
      throw new Error(
        'R2StorageAdapter non encore implémenté. ' +
        'Utiliser STORAGE_BACKEND=local en développement. ' +
        'L\'adaptateur R2 sera ajouté en Phase 6.'
      )

    default:
      throw new Error(
        `Valeur STORAGE_BACKEND non reconnue : "${backend}". ` +
        'Valeurs acceptées : "local", "r2".'
      )
  }
}

/**
 * LocalStorageAdapter — Adaptateur de stockage sur le système de fichiers local.
 *
 * Utilisé en développement (STORAGE_BACKEND=local).
 * Les fichiers sont stockés dans le dossier configuré par LOCAL_UPLOAD_DIR
 * (par défaut : "./uploads" à la racine du projet).
 *
 * Structure des dossiers gérée par cet adaptateur :
 *   uploads/
 *     originals/   ← images telles qu'uploadées (jamais écrasées)
 *     processed/   ← images traitées par Sharp
 *     thumbnails/  ← miniatures 400px
 *
 * IMPORTANT : Ne jamais importer cette classe directement dans le code applicatif.
 * Toujours utiliser getStorage() depuis @/lib/storage pour garantir
 * que le swap local→R2 (Phase 6) ne nécessite aucun changement de code.
 *
 * [M6 — PITFALLS.md] La DB stocke des clés, pas des chemins absolus.
 */

import * as fs from 'node:fs/promises'
import * as path from 'node:path'
import type { StorageProvider } from './index'

/**
 * Implémentation locale du StorageProvider.
 * Écrit et lit les fichiers sur le disque du serveur de développement.
 */
export class LocalStorageAdapter implements StorageProvider {
  /**
   * Dossier racine où les fichiers sont stockés.
   * Lu depuis LOCAL_UPLOAD_DIR (défaut : "./uploads" relatif au projet).
   */
  private readonly uploadDir: string

  /**
   * URL de base de l'application, utilisée pour construire les URLs publiques.
   * Lu depuis BETTER_AUTH_URL (ex: "http://localhost:3000").
   */
  private readonly baseUrl: string

  constructor() {
    // Résoudre le chemin en absolu par rapport à la racine du projet
    // process.cwd() retourne le dossier depuis lequel Next.js est lancé
    const uploadDirRaw = process.env['LOCAL_UPLOAD_DIR'] ?? './uploads'
    this.uploadDir = path.resolve(process.cwd(), uploadDirRaw)

    // Enlever le slash final s'il y en a un (évite les doubles slashes dans les URLs)
    this.baseUrl = (process.env['BETTER_AUTH_URL'] ?? 'http://localhost:3000').replace(/\/$/, '')
  }

  /**
   * Stocker un fichier sur le disque.
   *
   * Crée automatiquement les sous-dossiers manquants avant d'écrire.
   * Exemple : put("originals/abc123.jpg", buffer) crée uploads/originals/ si absent.
   *
   * @param key     Clé de stockage (ex: "originals/abc123.jpg")
   * @param buffer  Contenu binaire du fichier à stocker
   */
  async put(
    key: string,
    buffer: Buffer,
    _options?: { contentType?: string }
  ): Promise<void> {
    // Construire le chemin absolu sur le disque
    const filePath = path.join(this.uploadDir, key)

    // Créer les dossiers parents si nécessaires (ex: uploads/originals/)
    // recursive: true évite une erreur si le dossier existe déjà
    await fs.mkdir(path.dirname(filePath), { recursive: true })

    // Écrire le fichier
    await fs.writeFile(filePath, buffer)
  }

  /**
   * Récupérer un fichier depuis le disque.
   *
   * @param key  Clé de stockage
   * @returns    Buffer contenant les données du fichier
   * @throws     Error si le fichier n'existe pas
   */
  async get(key: string): Promise<Buffer> {
    const filePath = path.join(this.uploadDir, key)

    try {
      // Lire le fichier et retourner son contenu sous forme de Buffer
      const data = await fs.readFile(filePath)
      return Buffer.from(data)
    } catch {
      // Transformer l'erreur système en message compréhensible
      throw new Error(
        `Fichier introuvable dans le stockage local : "${key}". ` +
        `Chemin vérifié : ${filePath}`
      )
    }
  }

  /**
   * Supprimer un fichier du disque.
   *
   * Ne lève pas d'erreur si le fichier est absent — comportement intentionnel
   * pour permettre les suppressions idempotentes (ex: re-traitement d'image).
   *
   * @param key  Clé de stockage
   */
  async delete(key: string): Promise<void> {
    const filePath = path.join(this.uploadDir, key)

    try {
      await fs.unlink(filePath)
    } catch (error) {
      // Ignorer l'erreur "fichier absent" (ENOENT)
      // Relancer toute autre erreur (permissions, disque plein, etc.)
      if (
        error instanceof Error &&
        'code' in error &&
        (error as NodeJS.ErrnoException).code === 'ENOENT'
      ) {
        return // Fichier absent = suppression déjà effectuée, rien à faire
      }
      throw error
    }
  }

  /**
   * Obtenir l'URL publique d'un fichier.
   *
   * En développement, Next.js sert le dossier uploads/ via la route /uploads/*.
   * Attention : cette route de fichiers statiques doit être configurée dans
   * next.config.ts si les fichiers sont hors du dossier /public.
   *
   * Format retourné : "http://localhost:3000/uploads/originals/abc123.jpg"
   *
   * @param key  Clé de stockage (ex: "originals/abc123.jpg")
   * @returns    URL complète pour afficher le fichier dans le navigateur
   */
  getUrl(key: string): string {
    // Normaliser les séparateurs de chemin Windows → URLs utilisent toujours /
    const normalizedKey = key.replace(/\\/g, '/')
    return `${this.baseUrl}/uploads/${normalizedKey}`
  }
}
