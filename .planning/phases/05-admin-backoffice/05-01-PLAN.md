2. Créer `src/actions/artworkAdmin.ts` avec :
   ```typescript
   'use server'
   /**
    * Actions serveur pour la gestion admin des œuvres.
    * Toutes les actions vérifient la session artiste en première ligne (défense en profondeur).
    */
   import { auth } from '@/lib/auth'
   import { headers } from 'next/headers'
   import { redirect } from 'next/navigation'
   import { revalidatePath } from 'next/cache'
   import { db } from '@/db'
   import { artworks } from '@/db/schema'
   import { eq } from 'drizzle-orm'
   import { getStorage } from '@/lib/storage'
   import { z } from 'zod'

   // Schéma Zod pour valider le statut — correspond à artworkStatusEnum
   const statusSchema = z.enum(['available', 'sold', 'given', 'reserved'])

   /**
    * Change le statut d'une œuvre (disponible, vendue, donnée, réservée).
    * artworkId est bindé depuis le Server Component parent (jamais exposé en input client).
    * Sécurité : auth.api.getSession() en première ligne — T-05-01.
    */
   export async function updateArtworkStatus(
     artworkId: number,
     prevState: unknown,
     formData: FormData
   ): Promise<{ success: true } | { error: string }> {
     const session = await auth.api.getSession({ headers: await headers() })
     if (!session) return { error: 'Non autorisé.' }

     const parsed = statusSchema.safeParse(formData.get('status'))
     if (!parsed.success) return { error: 'Statut invalide.' }

     await db.update(artworks).set({ status: parsed.data }).where(eq(artworks.id, artworkId))
     revalidatePath(`/admin/artworks/${artworkId}`)
     revalidatePath('/admin/artworks')
     return { success: true }
   }

   /**
    * Supprime une œuvre : fichiers storage PUIS entrée DB (ordre critique).
    * La cascade FK supprime automatiquement les offres liées (onDelete: 'cascade').
    * Utilise redirect() — NE PAS appeler via useActionState, utiliser startTransition.
    * Sécurité : auth.api.getSession() en première ligne — T-05-01.
    */
   export async function deleteArtwork(artworkId: number): Promise<{ error: string } | never> {
     const session = await auth.api.getSession({ headers: await headers() })
     if (!session) return { error: 'Non autorisé.' }

     const artwork = await db.query.artworks.findFirst({ where: eq(artworks.id, artworkId) })
     if (!artwork) return { error: 'Œuvre introuvable.' }

     const storage = getStorage()
     // Ordre OBLIGATOIRE : storage d'abord, DB ensuite
     // .catch(() => {}) : idempotent — ignore ENOENT si le fichier est déjà absent
     if (artwork.originalPath) await storage.delete(artwork.originalPath).catch(() => {})
     if (artwork.processedPath) await storage.delete(artwork.processedPath).catch(() => {})
     if (artwork.thumbnailPath) await storage.delete(artwork.thumbnailPath).catch(() => {})

     // La cascade FK supprime automatiquement les offres et artwork_tags liés
     await db.delete(artworks).where(eq(artworks.id, artworkId))

     redirect('/admin/artworks')
   }
   ```

'use client'
/**
 * Formulaire de changement de statut d'une œuvre.
 *
 * POURQUOI un Client Component ?

import { useActionState } from 'react'

interface StatusChangeFormProps {
  currentStatus: 'available' | 'sold' | 'given' | 'reserved'
  updateStatusAction: (prevState: unknown, formData: FormData) => Promise<{ success: true } | { error: string }>
}

// Libellés affichés pour chaque statut — UI-SPEC.md contrat copywriting

'use client'
/**
 * Bouton de suppression d'œuvre avec confirmation AlertDialog.
 *
 * POURQUOI AlertDialog et pas window.confirm ?
 * window.confirm est bloquant et non stylisable. AlertDialog Radix est accessible,
 * stylisable, et suit D-03 (zéro border-radius). [D-05]
 *

 */
import { useTransition } from 'react'
import {
  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader,
  AlertDialogTitle, AlertDialogTrigger,
} from '@/components/ui/alert-dialog'

interface DeleteArtworkButtonProps {
  deleteAction: () => Promise<{ error: string } | never>
}

export function DeleteArtworkButton({ deleteAction }: DeleteArtworkButtonProps) {

    <AlertDialog>
      <AlertDialogTrigger asChild>
        <button
          disabled={isPending}

            height: '44px',
