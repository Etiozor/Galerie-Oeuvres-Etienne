**Étape 4 — Créer `src/actions/savedArtworks.ts`**

```typescript
'use server'
/**
 * Actions serveur pour la gestion des favoris visiteur (table saved_artworks).
 * Sécurité : vérification session visiteur en première ligne de chaque action.
 */
import { and, eq } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'
import { db } from '@/db'
import { savedArtworks } from '@/db/schema'
import { getVisitorSession } from '@/lib/session'

type ToggleResult =
  | { success: true; saved: boolean }
  | { success: false; errors: Record<string, string[]> }

/**
 * Ajoute ou retire une œuvre des favoris du visiteur connecté.
 *
 * IDEMPOTENCE : si le favori existe → supprime ; sinon → insère.
 * Pas d'upsert PostgreSQL : on a besoin de connaître l'état avant pour retourner
 * le bon résultat (saved: true/false) au Client Component.
 *
 * Sécurité : getVisitorSession() vérifié en première ligne — T-05-04.
 */
export async function toggleSavedArtwork(artworkId: number): Promise<ToggleResult> {
  const session = await getVisitorSession()

  if (!session.visitorId) {
    return { success: false, errors: { _form: ['Identifiez-vous pour enregistrer une œuvre.'] } }
  }

  const visitorId = session.visitorId

  try {
    // Vérifier si le favori existe déjà
    const existing = await db
      .select()
      .from(savedArtworks)
      .where(and(
        eq(savedArtworks.visitorId, visitorId),
        eq(savedArtworks.artworkId, artworkId),
      ))
      .limit(1)

    if (existing.length > 0) {
      // Retirer le favori
      await db.delete(savedArtworks)
        .where(and(
          eq(savedArtworks.visitorId, visitorId),
          eq(savedArtworks.artworkId, artworkId),
        ))
    } else {
      // Ajouter le favori
      await db.insert(savedArtworks).values({ visitorId, artworkId })
    }

    // Invalider le cache : page détail de l'œuvre + page profil
    revalidatePath(`/artworks/${artworkId}`)
    revalidatePath('/mon-profil')

    // saved: true si on vient d'ajouter (existing était vide)
    return { success: true, saved: existing.length === 0 }
  } catch (error) {
    console.error('[toggleSavedArtwork] Erreur serveur :', error)
    return { success: false, errors: { _form: ['Une erreur serveur est survenue. Veuillez réessayer.'] } }
  }
}
```

/**
 * Page profil du visiteur connecté — /mon-profil.
 *
 * Server Component : lit la session iron-session, charge les données depuis la DB.
 * SÉCURITÉ : si non connecté → redirect('/') immédiat (D-12).
 *
 * 4 sections (D-13) :
 * 1. Mes informations : displayName, téléphone masqué, email si fourni
 * 2. Mes offres : toutes les offres du visiteur, toutes œuvres confondues
 * 3. Mes acquisitions : heuristique D-18 (artwork.status === 'sold' + offre soumise)
 * 4. Mes œuvres enregistrées : table saved_artworks
 *
 * Typographie (UI-SPEC.md /mon-profil) :
 * - h1 : Oswald 28px/600 uppercase
 * - h2 sections : Space Mono 16px/400 uppercase letter-spacing 0.08em
 * - Corps : Geist Sans 16px/400
 * - Téléphone masqué : Space Mono 14px/400
 */
import { redirect } from 'next/navigation'
import { getVisitorSession } from '@/lib/session'
import { db } from '@/db'
import { artworks, offers, savedArtworks, visitors } from '@/db/schema'
import { eq, desc, and } from 'drizzle-orm'
import Link from 'next/link'
import { SiteNav } from '@/components/layout/SiteNav'

/**
 * Masque le numéro de téléphone : affiche "••••" + les 4 derniers chiffres.
 * Protège la vie privée du visiteur sur sa propre page de profil.
 */
function maskPhone(phone: string): string {
  return '••••' + phone.slice(-4)
}

export default async function MonProfilPage() {
  // ── 1. Guard session — JAMAIS session.save() ici (réservé aux Server Actions) ──
  const session = await getVisitorSession()
  if (!session.visitorId) {
    // Non connecté → redirection vers la galerie (D-12)
    redirect('/')
  }
  const visitorId = session.visitorId

  // ── 2. Charger les données du visiteur connecté ──
  const [visitorRows, myOffers, mySavedArtworks] = await Promise.all([
    // Informations du compte
    db.select().from(visitors).where(eq(visitors.id, visitorId)).limit(1),

    // Toutes les offres du visiteur — toutes œuvres confondues, triées par date (plus récente en tête)
    db.select({
      artworkId: artworks.id,
      artworkTitle: artworks.title,
      artworkStatus: artworks.status,
      amount: offers.amount,   // string (numeric Drizzle)
      message: offers.message,
      createdAt: offers.createdAt,
    })
    .from(offers)
    .leftJoin(artworks, eq(offers.artworkId, artworks.id))
    .where(eq(offers.visitorId, visitorId))
    .orderBy(desc(offers.createdAt)),

    // Favoris du visiteur
    db.select({ artwork: artworks })
    .from(savedArtworks)
    .leftJoin(artworks, eq(savedArtworks.artworkId, artworks.id))
    .where(eq(savedArtworks.visitorId, visitorId))
    .orderBy(desc(savedArtworks.createdAt)),
  ])

  const visitor = visitorRows[0]
  if (!visitor) redirect('/')

  // ── 3. Calculer les acquisitions probables (D-18) ──
  // Heuristique : œuvre vendue (status='sold') + visiteur a soumis au moins une offre dessus.
  // Dédoublonnage : si plusieurs offres sur la même œuvre sold → une seule acquisition.
  const myAcquisitions = myOffers
    .filter(o => o.artworkStatus === 'sold')
    .reduce<typeof myOffers>((acc, curr) => {
      if (!acc.find(a => a.artworkId === curr.artworkId)) acc.push(curr)
      return acc
    }, [])

  // ── Styles partagés ──
  const sectionStyle = { marginTop: '40px', paddingTop: '32px', borderTop: '1px solid rgba(0,0,0,0.1)' }
  const h2Style = {
    fontFamily: 'var(--font-space-mono)',
    fontSize: '16px',
    fontWeight: 400,
    textTransform: 'uppercase' as const,
    letterSpacing: '0.08em',
    color: 'var(--color-ink)',
    marginBottom: '20px',
    margin: 0,
  }

  return (
    <>
      <SiteNav />
      <main style={{ maxWidth: '720px', margin: '0 auto', padding: '48px 24px', fontFamily: 'var(--font-geist-sans)', backgroundColor: 'var(--color-bg)' }}>

        {/* ── En-tête : carré initiale + displayName ── */}
        <div style={{ display: 'flex', alignItems: 'center', gap: '16px', marginBottom: '40px' }}>
          {/* Carré initiale 40px — D-03 : pas de border-radius, fond ink, texte blanc (D-16) */}
          <div style={{
            width: '40px', height: '40px',
            backgroundColor: 'var(--color-ink)',
            display: 'flex', alignItems: 'center', justifyContent: 'center',
            flexShrink: 0,
          }}>
            <span style={{ fontFamily: 'var(--font-space-mono)', fontSize: '16px', fontWeight: 700, color: '#ffffff', lineHeight: 1 }}>
              {visitor.displayName[0]?.toUpperCase() ?? '?'}
            </span>
          </div>
          <h1 style={{ fontFamily: 'var(--font-oswald)', fontSize: '28px', fontWeight: 600, textTransform: 'uppercase', letterSpacing: '0.05em', color: 'var(--color-ink)', margin: 0 }}>
            {visitor.displayName}
          </h1>
        </div>

        {/* ── Section 1 : Mes informations ── */}
        <section>
          <h2 style={h2Style}>Mes informations</h2>

```typescript
'use client'
/**
 * Bouton cœur pour enregistrer/retirer une œuvre des favoris.
 *
 * POURQUOI un Client Component ?
 * L'état toggle (cœur plein / vide) est mis à jour optimistiquement côté client
 * via useState. La Server Action confirme en DB via startTransition.
 *
 * Visible uniquement si le visiteur est connecté (conditionnel dans la page parente — D-14).
 */
import { useState, useTransition } from 'react'
import { toggleSavedArtwork } from '@/actions/savedArtworks'

interface SaveButtonProps {
  artworkId: number
  initialSaved: boolean
}

export function SaveButton({ artworkId, initialSaved }: SaveButtonProps) {
  const [isSaved, setIsSaved] = useState(initialSaved)
  const [isPending, startTransition] = useTransition()

  function handleToggle() {
    // Mise à jour optimiste de l'état local
    setIsSaved(prev => !prev)
    startTransition(async () => {
      const result = await toggleSavedArtwork(artworkId)
      if (result.success) {
        // Synchroniser avec la réponse serveur (état canonique)
        setIsSaved(result.saved)
      } else {
        // En cas d'erreur, revenir à l'état initial
        setIsSaved(prev => !prev)
      }
    })
  }

  return (
    <button
      onClick={handleToggle}
      disabled={isPending}
      aria-label={isSaved ? 'Retirer des enregistrements' : 'Enregistrer l\'œuvre'}
      style={{
        width: '36px',
        height: '36px',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        backgroundColor: isSaved ? 'var(--color-ink)' : 'transparent',
        border: '1px solid var(--color-ink)',
        borderRadius: 0,  // D-03 : pas de border-radius
        cursor: isPending ? 'not-allowed' : 'pointer',
        opacity: isPending ? 0.6 : 1,
        transition: 'background-color 0.15s ease',
      }}
    >
      {/* ♥ plein si enregistré, ♡ vide sinon — Unicode évite une dépendance d'icône */}
      <span style={{
        fontSize: '16px',
        color: isSaved ? '#ffffff' : 'var(--color-ink)',
        lineHeight: 1,
        userSelect: 'none',
      }}>
        {isSaved ? '♥' : '♡'}
      </span>
    </button>
  )
}
```
