const result = await db
  .select({ total: count() })
  .from(offers)
  .where(and(eq(offers.visitorId, visitorId), eq(offers.artworkId, artworkId)))

| T-04-03-02 | Falsification | belowFloor calculé côté client | mitigate | belowFloor calculé dans submitOffer via basePrice récupéré depuis la DB (piège 5 RESEARCH.md) |
