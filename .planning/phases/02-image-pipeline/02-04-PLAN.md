    **Étape C — Créer src/actions/validation.ts (GREEN) :**

    ```typescript
    /**
     * Server Actions de validation pipeline — décisions finales de l'artiste.
     *
     * Trois actions correspondant aux trois boutons de l'écran de validation (D-04) :
     * 1. approveArtwork  → publie la version traitée par Sharp (PIPE-04)
     * 2. publishRaw      → publie l'original sans correction (PIPE-05)
     * 3. rejectArtwork   → supprime l'entrée et tous ses fichiers
     *
     * POURQUOI des Server Actions et pas des Route Handlers ?
     * Pattern établi du projet : toutes les mutations passent par des Server Actions.
     * La page de validation utilise <form action={serverAction.bind(null, id)}> — natif Next.js 15.
     */
    'use server'

    import { auth } from '@/lib/auth'
    import { headers } from 'next/headers'
    import { redirect } from 'next/navigation'
    import { db } from '@/db'
    import { artworks } from '@/db/schema'
    import { eq } from 'drizzle-orm'
    import { getStorage } from '@/lib/storage'

    /**
     * Approuve la version traitée par Sharp et la publie.
     * processedPath contient la version WebP corrigée → passe en 'published'.
     */
    export async function approveArtwork(
      artworkId: number
    ): Promise<{ error?: string }> {
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) return { error: 'Non autorisé.' }

      const artwork = await db.query.artworks.findFirst({
        where: eq(artworks.id, artworkId),
      })
      if (!artwork) return { error: 'Œuvre introuvable.' }

      // Publier la version traitée — publishedAt marque la date de mise en ligne
      await db.update(artworks)
        .set({
          pipelineState: 'published',
          publishedAt: new Date(),
        })
        .where(eq(artworks.id, artworkId))

      // Rediriger vers la liste après approbation
      redirect('/admin/artworks')
    }

    /**
     * Publie l'image originale sans correction (PIPE-05 — D-04 "Publier l'original").
     *
     * POURQUOI copier originalPath dans processedPath ?
     * La galerie publique (Phase 3) utilisera processedPath pour afficher les images.
     * En assignant originalPath → processedPath, on garantit que le même champ
     * est utilisé partout, qu'il y ait eu correction ou non.
     */
    export async function publishRaw(
      artworkId: number
    ): Promise<{ error?: string }> {
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) return { error: 'Non autorisé.' }

      const artwork = await db.query.artworks.findFirst({
        where: eq(artworks.id, artworkId),
      })
      if (!artwork) return { error: 'Œuvre introuvable.' }

      // PIPE-05 : l'original devient la version "traitée" dans la DB
      await db.update(artworks)
        .set({
          processedPath: artwork.originalPath, // l'original est maintenant le "traité"
          pipelineState: 'published',
          publishedAt: new Date(),
        })
        .where(eq(artworks.id, artworkId))

      redirect('/admin/artworks')
    }

    /**
     * Rejette l'œuvre : supprime les 3 fichiers stockés et supprime la ligne DB.
     *
     * POURQUOI supprimer les fichiers avant la ligne DB ?
     * Si la suppression DB échoue après les suppressions storage, on a des fichiers
     * orphelins mais la ligne DB pointe vers des clés qui n'existent plus.
     * Supprimer dans cet ordre permet un retry propre si la DB échoue.
     * Si les suppressions storage échouent, on laisse la ligne DB intacte
     * (les fichiers existent encore — cohérence maintenue).
     */
    export async function rejectArtwork(
      artworkId: number
    ): Promise<{ error?: string }> {
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) return { error: 'Non autorisé.' }

      const artwork = await db.query.artworks.findFirst({
        where: eq(artworks.id, artworkId),
      })
      if (!artwork) return { error: 'Œuvre introuvable.' }

      const storage = getStorage()

      // Supprimer les 3 variantes stockées — silencieux si absentes (idempotent)
      if (artwork.originalPath) {
        await storage.delete(artwork.originalPath).catch(() => {})
      }
      if (artwork.processedPath) {
        await storage.delete(artwork.processedPath).catch(() => {})
      }
      if (artwork.thumbnailPath) {
        await storage.delete(artwork.thumbnailPath).catch(() => {})
      }

      // Supprimer la ligne DB après les fichiers
      await db.delete(artworks).where(eq(artworks.id, artworkId))

      redirect('/admin/artworks')
    }
    ```

    /**
     * Page de validation pipeline — /admin/artworks/[id]/validate.
     *
     * POURQUOI Server Component pour le parent ?
     * Les images et les données de l'œuvre viennent de la DB — rendu serveur optimal.
     *
     * POURQUOI un Client Component enfant pour les boutons ?
     * Les boutons doivent gérer un état de chargement (isPending) et
     * la modale AlertDialog (état ouvert/fermé) — nécessitent 'use client'.
     *
     * Pattern : Server Component passe les URLs et les actions au Client Component.
     */
    import { auth } from '@/lib/auth'
    import { headers } from 'next/headers'
    import { redirect, notFound } from 'next/navigation'
    import { db } from '@/db'
    import { artworks } from '@/db/schema'
    import { eq } from 'drizzle-orm'
    import { getStorage } from '@/lib/storage'
    import { approveArtwork, publishRaw, rejectArtwork } from '@/actions/validation'
    import { ValidationActions } from './_components/ValidationActions'

    export default async function ValidatePage({
      params,
    }: {
      params: Promise<{ id: string }>
    }) {
      // Next.js 15 : params est une Promise
      const { id } = await params
      const artworkId = parseInt(id, 10)

      // Auth
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) redirect('/admin/login')

      // Récupérer l'œuvre
      const artwork = await db.query.artworks.findFirst({
        where: eq(artworks.id, artworkId),
      })
      if (!artwork) notFound()

      // Rediriger si l'état n'est pas AWAITING_VALIDATION
      // (évite d'afficher l'écran de validation pour une œuvre déjà publiée ou en erreur)
      if (artwork.pipelineState !== 'awaiting_validation') {
        redirect('/admin/artworks')
      }

      // Obtenir les URLs des deux images (servies par /api/uploads/[...key])
      const storage = getStorage()
      const originalUrl = artwork.originalPath ? storage.getUrl(artwork.originalPath) : null
      const processedUrl = artwork.processedPath ? storage.getUrl(artwork.processedPath) : null

      // Binder l'ID aux Server Actions pour les formulaires
      const approveWithId = approveArtwork.bind(null, artwork.id)
      const publishRawWithId = publishRaw.bind(null, artwork.id)
      const rejectWithId = rejectArtwork.bind(null, artwork.id)

      return (
        <main style={{ padding: '48px 16px', fontFamily: 'var(--font-geist-sans)' }}>
          {/* Titre de page — Display 28px / 600 (UI-SPEC) */}
          <h1 style={{ fontSize: '28px', fontWeight: 600, marginBottom: '32px' }}>
            Valider la correction
          </h1>

          {/* Comparaison côte à côte — D-03 : original gauche, corrigé droite */}
          <div
            style={{
              display: 'grid',
              gridTemplateColumns: 'repeat(auto-fit, minmax(280px, 1fr))',
              gap: '32px', // gap-xl (UI-SPEC spacing xl = 32px)
              marginBottom: '32px',
              backgroundColor: '#f5f5f5', // fond secondaire UI-SPEC
              padding: '24px',
              borderRadius: '8px',
            }}
          >
            {/* Colonne gauche : Original */}
            <div>
              <p style={{ fontSize: '14px', fontWeight: 600, marginBottom: '8px', color: '#171717' }}>
                Original
              </p>
              {originalUrl ? (
                <img
                  src={originalUrl}
                  alt="Image originale"
                  aria-label="Image originale avant correction"
                  style={{ width: '100%', height: 'auto', borderRadius: '4px', display: 'block' }}
                />
              ) : (
                <div style={{ padding: '24px', color: '#737373', textAlign: 'center' }}>
                  Image originale non disponible
                </div>
              )}
            </div>

            {/* Colonne droite : Version corrigée */}
            <div>
              <p style={{ fontSize: '14px', fontWeight: 600, marginBottom: '8px', color: '#171717' }}>
                Version corrigée
              </p>
              {processedUrl ? (
                <img
                  src={processedUrl}
                  alt="Image après correction par le pipeline"
                  aria-label="Image corrigée par le pipeline (rotation EXIF, normalisation, balance des blancs)"
                  style={{ width: '100%', height: 'auto', borderRadius: '4px', display: 'block' }}
                />
              ) : (
                <div style={{ padding: '24px', color: '#737373', textAlign: 'center' }}>
                  Image traitée non disponible
                </div>
              )}
            </div>
          </div>

          {/* Actions de validation — Client Component pour gérer loading + modale */}
          <ValidationActions
            approveAction={approveWithId}
            publishRawAction={publishRawWithId}
            rejectAction={rejectWithId}
          />
        </main>
      )
    }

    'use client'

    import { useTransition, useState } from 'react'
    import { Button } from '@/components/ui/button'
    import {
      AlertDialog,
      AlertDialogAction,
      AlertDialogCancel,
      AlertDialogContent,
      AlertDialogDescription,
      AlertDialogFooter,
      AlertDialogHeader,
      AlertDialogTitle,
      AlertDialogTrigger,
    } from '@/components/ui/alert-dialog'

    interface ValidationActionsProps {
      approveAction: () => Promise<{ error?: string } | void>
      publishRawAction: () => Promise<{ error?: string } | void>
      rejectAction: () => Promise<{ error?: string } | void>
    }

    export function ValidationActions({
      approveAction,
      publishRawAction,
      rejectAction,
    }: ValidationActionsProps) {
      const [isPending, startTransition] = useTransition()
      const [activeAction, setActiveAction] = useState<'approve' | 'raw' | 'reject' | null>(null)

      function handleApprove() {
        setActiveAction('approve')
        startTransition(async () => {
          await approveAction()
          setActiveAction(null)
        })

        setActiveAction('raw')
        startTransition(async () => {
          await publishRawAction()
          setActiveAction(null)
        })
      }

      function handleReject() {
        setActiveAction('reject')
        startTransition(async () => {
          await rejectAction()
          setActiveAction(null)
        })
      }

      return (
        <div
          style={{
            display: 'flex',
            flexDirection: 'column',
            gap: '12px',
            maxWidth: '400px',
          }}
        >
          {/* Bouton 1 — CTA primaire : Approuver la correction (UI-SPEC) */}
          <Button
            onClick={handleApprove}
            disabled={isPending}
            aria-label="Approuver la correction et publier la version améliorée par le pipeline"
