    ```typescript
    /**
     * Server Action d'upload d'œuvre — point d'entrée du pipeline image.
     *
     * POURQUOI ici et pas dans une API Route ?
     * Les Server Actions sont le pattern établi de ce projet pour toutes les mutations.
     * Pas besoin d'un endpoint REST séparé — Next.js gère le FormData directement.
     *
     * Flux : Validation → DB INSERT (PENDING) → StorageProvider.put(original)
     * Le pipeline Sharp est déclenché dans le plan 02-02.
     */
    'use server'

    import { auth } from '@/lib/auth'
    import { headers } from 'next/headers'
    import { db } from '@/db'
    import { artworks } from '@/db/schema'
    import { eq } from 'drizzle-orm'
    import { getStorage } from '@/lib/storage'
    import { randomUUID } from 'crypto'

    const MAX_SIZE_BYTES = 20 * 1024 * 1024 // 20 MB — D-07
    const ALLOWED_TYPES = ['image/jpeg', 'image/png'] // D-06
    const HEIC_TYPES = ['image/heic', 'image/heif'] // D-06

    /**
     * Uploade une photo et crée une entrée en base avec l'état PENDING.
     *
     * POURQUOI créer l'entrée DB avant le traitement Sharp ?
     * L'artiste a décidé (D-01, D-02) de voir l'image apparaître dans
     * sa liste immédiatement. Le traitement est asynchrone.
     *
     * @param _prevState  État précédent (requis par useActionState React 19)
     * @param formData    FormData contenant le champ 'file'
     */
    export async function uploadArtwork(
      _prevState: { error?: string; success?: boolean; artworkId?: number } | null,
      formData: FormData
    ): Promise<{ error?: string; success?: boolean; artworkId?: number }> {
      // Vérifier que l'artiste est connecté avant tout traitement
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) return { error: 'Non autorisé.' }

      const file = formData.get('file')
      if (!(file instanceof File) || file.size === 0) {
        return { error: 'Aucun fichier reçu.' }
      }

      // Validation HEIC en premier (message spécifique pour les iPhones — D-06)
      if (HEIC_TYPES.includes(file.type)) {
        return {
          error: "Les photos iPhone en format HEIC ne sont pas supportées. Exporte en JPEG depuis l'app Photos.",
        }
      }

      // Validation format — uniquement JPEG et PNG (D-06)
      if (!ALLOWED_TYPES.includes(file.type)) {
        return {
          error: "Ce format de fichier n'est pas supporté. Utilise un fichier JPEG ou PNG.",
        }
      }

      // Validation taille — max 20 MB (D-07)
      if (file.size > MAX_SIZE_BYTES) {
        return {
          error: "Le fichier est trop lourd (max 20 MB). Compresse-le avant d'uploader.",
        }
      }

      // Convertir File → Buffer pour le StorageProvider
      const buffer = Buffer.from(new Uint8Array(await file.arrayBuffer()))
      const ext = file.type === 'image/png' ? 'png' : 'jpg'
      const uuid = randomUUID()
      const originalKey = `originals/${uuid}.${ext}`
      const storage = getStorage()

      // Créer l'entrée DB en PENDING immédiatement (D-02)
      // titre = 'Sans titre' par défaut (D-01 : métadonnées remplies après upload)
      const [artwork] = await db
        .insert(artworks)
        .values({
          title: 'Sans titre',
          pipelineState: 'pending',
          originalPath: originalKey,
        })
        .returning({ id: artworks.id })

      // Stocker le fichier original — JAMAIS écrasé (D-13)
      await storage.put(originalKey, buffer, { contentType: file.type })

      return { success: true, artworkId: artwork.id }
    }
    ```

    ```typescript
    /**
     * Route Handler pour servir les fichiers uploadés depuis le stockage local.
     *
     * POURQUOI une route API et pas le dossier /public ?
     * Les fichiers uploadés sont écrits au runtime dans ./uploads/ (hors de /public).
     * Next.js ne sert pas les fichiers hors /public comme assets statiques.
     * En production R2 (Phase 6), ce handler peut être ignoré — R2 sert ses propres URLs CDN.
     *
     * LocalStorageAdapter.getUrl("originals/abc.jpg") retourne
     * "http://localhost:3000/uploads/originals/abc.jpg" — ce handler répond à ces URLs.
     */
    import { type NextRequest, NextResponse } from 'next/server'
    import { getStorage } from '@/lib/storage'

    export async function GET(
      _request: NextRequest,
      { params }: { params: Promise<{ key: string[] }> }
    ) {
      // Next.js 15 : params est une Promise, doit être await-ée
      const { key } = await params
      const storageKey = key.join('/')

      try {
        const storage = getStorage()
        const buffer = await storage.get(storageKey)
        const contentType = detectContentType(storageKey)
        return new Response(buffer, {
          headers: {
            'Content-Type': contentType,
            // Cache 1 heure — les images uploadées ne changent pas
            'Cache-Control': 'public, max-age=3600',
          },
        })
      } catch {
        // Fichier absent ou clé invalide
        return new NextResponse('Fichier introuvable', { status: 404 })
      }
    }

    /**
     * Détermine le Content-Type depuis l'extension de la clé de stockage.
     * POURQUOI ne pas lire le magic bytes ? La clé contient toujours l'extension
     * correcte car on la construit nous-mêmes dans uploadArtwork.
     */
    function detectContentType(key: string): string {
      if (key.endsWith('.webp')) return 'image/webp'
      if (key.endsWith('.jpg') || key.endsWith('.jpeg')) return 'image/jpeg'
      if (key.endsWith('.png')) return 'image/png'
      return 'application/octet-stream'
    }
    ```
