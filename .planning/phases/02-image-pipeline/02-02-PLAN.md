    /**
     * Pipeline de traitement d'image Sharp — Galerie POF.
     *
     * POURQUOI cette fonction est séparée de la Server Action ?
     * 1. Testabilité : une fonction pure Buffer → Buffer peut être testée
     *    sans Next.js, sans DB, sans StorageProvider.
     * 2. Réutilisabilité : l'action "Réessayer" (D-09) appelle la même
     *    fonction avec l'original stocké, sans dupliquer la logique Sharp.
     *
     * ORDRE OBLIGATOIRE des opérations Sharp :
     * .rotate() TOUJOURS en premier (D-11) — corrige l'orientation EXIF avant
     * que normalise() analyse les pixels. Si rotate() vient après, l'algorithme
     * d'exposition travaille sur des pixels mal orientés.
     *
     * .withMetadata() TOUJOURS présent (D-12) — préserve le profil ICC Display-P3
     * des iPhone récents. Sans lui, Sharp strip le profil et les couleurs paraissent
     * désaturées dans les navigateurs à gestion des couleurs.
     */
    import sharp from 'sharp'

    /**
     * Résultat du pipeline de traitement.
     */
    export interface PipelineResult {
      /** Image pleine résolution en WebP (qualité 85), corrigée et ICC préservé */
      processedBuffer: Buffer
      /** Miniature 400px max en WebP (qualité 80), même pipeline de correction */
      thumbnailBuffer: Buffer
    }

    /**
     * Traite une image avec Sharp pour produire deux variantes WebP.
     *
     * @param inputBuffer  Buffer de l'image originale (JPEG ou PNG)
     * @returns            { processedBuffer, thumbnailBuffer }
     * @throws             Error si Sharp ne peut pas traiter le buffer (format invalide, corrompu)
     */
    export async function processImage(inputBuffer: Buffer): Promise<PipelineResult> {
      // --- Version traitée — pleine résolution ---
      // POURQUOI cet ordre exact ?
      // 1. rotate() : lit le tag EXIF Orientation et réoriente les pixels PHYSIQUEMENT.
      //    Doit être en premier car toutes les opérations suivantes supposent une image bien orientée.
      // 2. normalise() : étire l'histogramme de luminosité pour corriger la sous/surexposition.
      //    lower:1 / upper:99 = ignore 1% des pixels les plus sombres et les plus clairs (PIPE-02).
      //    PIPE-03 : normalise() travaille par canal (R, G, B séparément) — cela effectue
      //    implicitement une balance des blancs en normalisant chaque canal indépendamment.
      // 3. withMetadata() : copie les métadonnées EXIF (incluant le profil ICC) dans l'output WebP (D-12).
      // 4. webp({ quality: 85 }) : encode en WebP, bon compromis qualité/taille pour les œuvres d'art.
      const processedBuffer = await sharp(inputBuffer)
        .rotate()                           // D-11 : EXIF rotation EN PREMIER (obligatoire)
        .normalise({ lower: 1, upper: 99 }) // PIPE-02/03 : exposition + contraste + balance couleurs
        .withMetadata()                     // D-12 : préserver profil ICC Display-P3 iPhone
        .webp({ quality: 85 })             // D-14 : sortie WebP qualité 85
        .toBuffer()

      // --- Miniature — largeur 400px maximum ---
      // Même pipeline de correction que la version traitée.
      // resize(400, null) = largeur fixe, hauteur proportionnelle.
      // withoutEnlargement: true = jamais agrandir une image déjà petite (évite les artefacts).
      const thumbnailBuffer = await sharp(inputBuffer)
        .rotate()                                              // D-11 : toujours en premier
        .resize(400, null, { withoutEnlargement: true })       // largeur 400px max (Claude's Discretion)
        .normalise({ lower: 1, upper: 99 })                   // même correction que la version traitée
        .withMetadata()                                        // D-12 : ICC préservé dans la miniature aussi
        .webp({ quality: 80 })                                // qualité légèrement réduite pour la miniature
        .toBuffer()

      return { processedBuffer, thumbnailBuffer }
    }
    ```

    /**
     * Server Action d'upload d'œuvre — pipeline complet.
     *
     * Flux complet (depuis RESEARCH.md Architecture Patterns) :
     * 1. Validation session, MIME, taille (PIPE-01)
     * 2. DB INSERT avec pipelineState='pending' (D-02)
     * 3. StorageProvider.put(originalKey) — original jamais écrasé (D-13)
     * 4. DB UPDATE pipelineState='processing'
     * 5. Sharp pipeline : rotate() → normalise() → withMetadata() → webp() (D-11, D-12, PIPE-02/03)
     * 6. StorageProvider.put(processedKey)
     * 7. Sharp thumbnail : rotate() → resize(400) → normalise() → withMetadata() → webp()
     * 8. StorageProvider.put(thumbnailKey)
     * 9a. Succès → DB UPDATE pipelineState='awaiting_validation', processedPath, thumbnailPath
     * 9b. Échec (catch) → DB UPDATE pipelineState='failed' + delete fichiers partiels (D-08)
     */
    'use server'

    import { auth } from '@/lib/auth'
    import { headers } from 'next/headers'
    import { db } from '@/db'
    import { artworks } from '@/db/schema'
    import { eq } from 'drizzle-orm'
    import { getStorage } from '@/lib/storage'
    import { processImage } from '@/lib/pipeline/process'
    import { randomUUID } from 'crypto'

    const MAX_SIZE_BYTES = 20 * 1024 * 1024 // 20 MB — D-07
    const ALLOWED_TYPES = ['image/jpeg', 'image/png'] // D-06
    const HEIC_TYPES = ['image/heic', 'image/heif'] // D-06

    export async function uploadArtwork(
      _prevState: { error?: string; success?: boolean; artworkId?: number } | null,
      formData: FormData
    ): Promise<{ error?: string; success?: boolean; artworkId?: number }> {
      // --- Étape 1 : Validation ---
      const session = await auth.api.getSession({ headers: await headers() })
      if (!session) return { error: 'Non autorisé.' }

      const file = formData.get('file')
      if (!(file instanceof File) || file.size === 0) {
        return { error: 'Aucun fichier reçu.' }
      }

      if (HEIC_TYPES.includes(file.type)) {
        return {
          error: "Les photos iPhone en format HEIC ne sont pas supportées. Exporte en JPEG depuis l'app Photos.",
        }
      }
      if (!ALLOWED_TYPES.includes(file.type)) {
        return {
          error: "Ce format de fichier n'est pas supporté. Utilise un fichier JPEG ou PNG.",
        }
      }
      if (file.size > MAX_SIZE_BYTES) {
        return {
          error: "Le fichier est trop lourd (max 20 MB). Compresse-le avant d'uploader.",
        }
      }

      // Convertir File → Buffer pour Sharp et StorageProvider
      const buffer = Buffer.from(new Uint8Array(await file.arrayBuffer()))
      const ext = file.type === 'image/png' ? 'png' : 'jpg'
      const uuid = randomUUID()
      const originalKey = `originals/${uuid}.${ext}`
      const processedKey = `processed/${uuid}.webp`
      const thumbnailKey = `thumbnails/${uuid}.webp`
      const storage = getStorage()

      // --- Étape 2 : DB INSERT en PENDING (D-02) ---
      const [artwork] = await db
        .insert(artworks)
        .values({
          title: 'Sans titre',
          pipelineState: 'pending',
          originalPath: originalKey,
        })
        .returning({ id: artworks.id })

      // --- Étape 3 : Stocker l'original — JAMAIS écrasé (D-13) ---
      await storage.put(originalKey, buffer, { contentType: file.type })

      // --- Étape 4 : Passer en PROCESSING ---
      await db.update(artworks)
        .set({ pipelineState: 'processing' })
        .where(eq(artworks.id, artwork.id))

      // --- Étapes 5–9 : Pipeline Sharp synchrone ---
      try {
        // Étapes 5–8 : Sharp génère les deux variantes (voir src/lib/pipeline/process.ts)
        const { processedBuffer, thumbnailBuffer } = await processImage(buffer)

        // Stocker les deux variantes WebP
        await storage.put(processedKey, processedBuffer, { contentType: 'image/webp' })
        await storage.put(thumbnailKey, thumbnailBuffer, { contentType: 'image/webp' })

        // Étape 9a : Succès → AWAITING_VALIDATION + enregistrer les chemins (D-13)
        await db.update(artworks)
          .set({
            pipelineState: 'awaiting_validation',
            processedPath: processedKey,
            thumbnailPath: thumbnailKey,
          })
          .where(eq(artworks.id, artwork.id))

        return { success: true, artworkId: artwork.id }
      } catch {
        // Étape 9b : Échec → FAILED (D-08)
        // POURQUOI ne pas rethrow ? La Server Action doit retourner une valeur,
        // pas lever une exception — sinon Next.js affiche une page d'erreur générique.
        await db.update(artworks)
          .set({ pipelineState: 'failed' })
          .where(eq(artworks.id, artwork.id))

        // Nettoyer les fichiers partiels — silencieux si absents (idempotent)
        // POURQUOI .catch(() => {}) ? Si seul processed a été écrit avant le crash,
        // la suppression de thumbnailKey échouerait silencieusement — acceptable.
        await storage.delete(processedKey).catch(() => {})
        await storage.delete(thumbnailKey).catch(() => {})

        return {
          error: 'Le traitement a échoué. Utilisez "Réessayer le traitement" dans la liste.',
        }
      }
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Server Action → Sharp | Buffer inconnu vient du client — Sharp peut planter sur contenu invalide |
| Sharp → StorageProvider | Les buffers produits par Sharp sont de confiance (générés par libvips) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-06 | Denial of Service | processImage — pixel flood | accept | La limite 20 MB upstream + les limites internes libvips (mémoire par pixel) suffisent pour v1. Un buffer 20 MB JPEG = ~60 MP max — libvips gère sans problème. |
| T-02-07 | Denial of Service | processImage — Sharp timeout | accept | RESEARCH.md Assumption A2 : Sharp traite 20 MB en < 10s. Si timeout survient, l'état FAILED est set dans le catch — pas de stuck PROCESSING. |
| T-02-08 | Tampering | processImage — contenu malicieux dans EXIF | accept | Sharp/libvips parse l'EXIF lui-même. CVEs connus dans libvips sont patchés rapidement (Sharp 0.34.5 = latest). `withMetadata()` copie l'EXIF tel quel — les métadonnées ne sont jamais exécutées. |
| T-02-09 | Tampering | Stuck PROCESSING si catch ne se déclenche pas | mitigate | Le catch est au niveau du bloc try complet — couvre Sharp + les deux storage.put(). Si le process Node.js crash, l'état reste PROCESSING — à gérer en Phase 5 (cleanup job). Acceptable pour v1 (artiste unique). |
</threat_model>
